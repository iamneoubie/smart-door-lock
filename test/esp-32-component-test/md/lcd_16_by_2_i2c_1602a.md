#include <Wire.h>
#include <LiquidCrystal_I2C.h>

// ============================================================
// PIN CONFIGURATION
// ============================================================
// LCD I2C Pins (ESP32)
// SDA -> GPIO21
// SCL -> GPIO22
// VCC -> 5V
// GND -> GND
// ============================================================

// LCD Configuration
#define LCD_ADDRESS 0x27 // Common I2C address, try 0x3F if this doesn't work
#define LCD_COLUMNS 16
#define LCD_ROWS 2

// Create LCD object
LiquidCrystal_I2C lcd(LCD_ADDRESS, LCD_COLUMNS, LCD_ROWS);

// Test timing variables (adjusted for ESP32 speed)
const unsigned long TEST_DELAY_SHORT = 1500;  // Increased from 1000
const unsigned long TEST_DELAY_MEDIUM = 2000; // Increased from 1500
const unsigned long TEST_DELAY_LONG = 2500;   // Increased from 2000
const unsigned long ANIMATION_DELAY = 150;    // Slower animation
const unsigned long SCROLL_DELAY = 300;       // Slower scrolling

// ============================================================
// HELPER FUNCTIONS
// ============================================================

/**
 * Improved wait function with Serial feedback
 */
void waitTest(unsigned long ms, const char *message = nullptr)
{
    if (message)
    {
        Serial.print("  [WAIT] ");
        Serial.print(ms);
        Serial.print("ms - ");
        Serial.println(message);
    }

    // Use millis() for non-blocking wait (better than delay)
    unsigned long startTime = millis();
    while (millis() - startTime < ms)
    {
        // Allow other tasks to run
        yield();
        delay(1);
    }
}

/**
 * Display test header on LCD and Serial
 */
void showTestHeader(const char *testName, int testNumber)
{
    Serial.println("----------------------------------------");
    Serial.print("TEST ");
    Serial.print(testNumber);
    Serial.print(": ");
    Serial.println(testName);

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Test ");
    lcd.print(testNumber);
    lcd.print(":");
    lcd.setCursor(0, 1);
    lcd.print(testName);

    // Give time for LCD to update
    delay(800);
}

/**
 * Update LCD with proper timing
 */
void updateLCD(int row, int col, const char *text)
{
    lcd.setCursor(col, row);
    lcd.print(text);
    delay(ANIMATION_DELAY / 3);
    yield(); // Allow ESP32 to handle other tasks
}

void testBasicText()
{
    showTestHeader("Basic Text", 1);

    lcd.clear();

    lcd.setCursor(0, 1);
    lcd.print("LCD 1602A Test");

    Serial.println("  [PASS] Displayed 'Hello World!' on both rows");
    waitTest(TEST_DELAY_LONG, "Displaying basic text");
}

// TEST 9: Custom Characters
// ============================================================
// Custom character definitions
byte heart[] = {
    0b00000,
    0b01010,
    0b11111,
    0b11111,
    0b01110,
    0b00100,
    0b00000,
    0b00000};

byte smiley[] = {
    0b00000,
    0b10001,
    0b00000,
    0b00000,
    0b10001,
    0b01110,
    0b00000,
    0b00000};

byte arrow[] = {
    0b00000,
    0b00100,
    0b01110,
    0b11111,
    0b01110,
    0b00100,
    0b00000,
    0b00000};

void testCustomCharacters()
{
    showTestHeader("Custom Characters", 9);

    // Create custom characters
    lcd.createChar(0, heart);
    lcd.createChar(1, smiley);
    lcd.createChar(2, arrow);

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Custom:");

    for (int col = 0; col < 8; col++)
    {
        lcd.setCursor(col + 7, 0);
        if (col % 3 == 0)
        {
            lcd.write(byte(0)); // Heart
            Serial.print("<3");
        }
        else if (col % 3 == 1)
        {
            lcd.write(byte(1)); // Smiley
            Serial.print(":)");
        }
        else
        {
            lcd.write(byte(2)); // Arrow
            Serial.print("^");
        }
        delay(ANIMATION_DELAY);
        yield();
    }

    lcd.setCursor(0, 1);
    lcd.print("CGRAM TEST OK");

    Serial.println("\n  [PASS] Custom characters created and displayed");
    waitTest(TEST_DELAY_LONG, "Displaying custom characters");
}

// ============================================================
// SETUP
// ============================================================
void setup()
{
    // Delay for Serial monitor connection
    delay(1000);

    // Initialize Serial
    Serial.begin(115200);
    delay(100);

    Serial.println("\n========================================");
    Serial.println("   ESP32 LCD 1602A I2C TEST SUITE");
    Serial.println("========================================");
    Serial.println("Pin Configuration:");
    Serial.println("  SDA -> GPIO21");
    Serial.println("  SCL -> GPIO22");
    Serial.println("  Address: 0x27 (16x2)");
    Serial.println("========================================\n");

    // Initialize I2C with slower speed for stability
    Wire.begin();
    Wire.setClock(100000); // 100kHz instead of 400kHz for better stability

    // Initialize LCD
    lcd.begin();
    lcd.backlight();
    lcd.clear();

    // Display startup message
    lcd.setCursor(0, 0);
    lcd.print("LCD 1602A");
    lcd.setCursor(0, 1);
    lcd.print("Starting Test");

    Serial.println("[INIT] LCD initialized successfully!");
    Serial.println("[INIT] Starting test sequence...\n");

    delay(2000);
}

// ============================================================
// LOOP - Run all tests
// ============================================================
void loop()
{
    // Run all tests in sequence with proper delays
    testBasicText();
    testHorizontalTop();
    testHorizontalBottom();
    testHorizontalBoth();
    testVertical();
    testPositions();
    testEveryPosition();
    testFullBlocks();
    testCustomCharacters();
    testCursor();
    testAlternatingRows();
    testScrolling();
    testBacklight();
    testDisplayControl();
    finalTest();

    // Wait before repeating
    Serial.println("\n[INFO] Test complete - Restarting in 5 seconds...\n");
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Test Complete");
    lcd.setCursor(0, 1);
    lcd.print("Restarting...");
    delay(5000);
}
