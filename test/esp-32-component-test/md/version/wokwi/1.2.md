/*
 * ESP32 Smart Lock System - VERSION 2.12
 * ======================================
 * Changes from v2.11:
 * - clearFirebaseState() no longer touches /commands/request.
 *   Only /sensors is cleared, and only when a first successful
 *   write confirms Firebase is actually reachable.
 * - HTTP helpers now use a 3-second timeout (default is 75s, which
 *   starved the keypad and made A/B/# unresponsive when Firebase
 *   was slow or offline).
 * - Keypad is polled FIRST in loop() so a slow Firebase call can
 *   never eat a key press.
 * - Explicit 'online' flag. Every Firebase call checks it first
 *   and returns instantly if offline.
 * - Backdoor PIN 54321 works fully offline: lock, unlock, status.
 * - # (temperature) always works offline.
 * - Firebase command poll only runs when online; if offline it
 *   does nothing and costs zero blocking time.
 */

String getTimestamp() {
    unsigned long elapsed = millis() - systemStartTime;
    unsigned long seconds = elapsed / 1000;
    unsigned long minutes = seconds / 60;
    unsigned long hours   = minutes / 60;
    seconds %= 60;
    minutes %= 60;
    char timeStr[20];
    snprintf(timeStr, sizeof(timeStr), "%02lu:%02lu:%02lu", hours, minutes, seconds);
    return String(timeStr);
}

void logMsg(String msg) {
    Serial.println("[" + getTimestamp() + "] " + msg);
}

// ============================================================
// RAW HTTP FIREBASE HELPERS
// ------------------------------------------------------------
// Every call short-circuits if 'online' is false, so the keypad
// loop is never blocked by a dead network. Timeout is 3s so even
// a slow link cannot starve input for more than 3 seconds.
// ============================================================

bool firebasePut(const String &path, const String &value) {
    if (!online) return false;
    HTTPClient http;
    http.setTimeout(HTTP_TIMEOUT_MS);
    String url = String(DATABASE_URL) + "/" + path + ".json?auth=" + DATABASE_AUTH;
    http.begin(url);
    http.addHeader("Content-Type", "application/json");
    int code = http.PUT(value);
    http.end();
    return (code > 0);
}

String firebaseGet(const String &path) {
    if (!online) return "";
    HTTPClient http;
    http.setTimeout(HTTP_TIMEOUT_MS);
    String url = String(DATABASE_URL) + "/" + path + ".json?auth=" + DATABASE_AUTH;
    http.begin(url);
    int code = http.GET();
    String payload = "";
    if (code > 0) {
        payload = http.getString();
        payload.trim();
        if (payload.startsWith("\"") && payload.endsWith("\"")) {
            payload = payload.substring(1, payload.length() - 1);
        }
    }
    http.end();
    return payload;
}

bool firebaseDelete(const String &path) {
    if (!online) return false;
    HTTPClient http;
    http.setTimeout(HTTP_TIMEOUT_MS);
    String url = String(DATABASE_URL) + "/" + path + ".json?auth=" + DATABASE_AUTH;
    http.begin(url);
    int code = http.sendRequest("DELETE");
    http.end();
    return (code > 0);
}

void setLCD(int row, const String &text) {
    String trimmed = text;
    if (trimmed.length() > 16) trimmed = trimmed.substring(0, 16);
    while (trimmed.length() < 16) trimmed += " ";

    if (row == 0) {
        if (lcdRow0 != trimmed) { lcdRow0 = trimmed; lcdNeedsUpdate = true; }
    } else {
        if (lcdRow1 != trimmed) { lcdRow1 = trimmed; lcdNeedsUpdate = true; }
    }
}

void updateLCD() {
    if (!lcdNeedsUpdate) return;
    if (millis() - lastLCDUpdateTime < LCD_UPDATE_INTERVAL) return;
    lastLCDUpdateTime = millis();
    lcd.setCursor(0, 0); lcd.print(lcdRow0);
    lcd.setCursor(0, 1); lcd.print(lcdRow1);
    lcdNeedsUpdate = false;
}

void showMainScreen() {
    currentDisplayMode = DISPLAY_MAIN;
    String status = isLocked ? "LOCKED" : "UNLOCKED";
    setLCD(0, status);
    setLCD(1, "A=LOCK B=UNLOCK");
    updateLCD();
}

    } else {
        display = pinBuffer;

    while (display.length() < 16) display += " ";
    setLCD(1, display);
    updateLCD();
}

void showTemperature() {
    currentDisplayMode = DISPLAY_TEMPERATURE;
    displayStartTime = millis();

    float h = dht.readHumidity();
    float t = dht.readTemperature();

    if (!isnan(h) && !isnan(t)) {
        lastTemperature = t; lastHumidity = h; hasDHTData = true;
        char line1[17], line2[17];
        snprintf(line1, sizeof(line1), "TEMP: %4.1f C", t);
        snprintf(line2, sizeof(line2), "HUM:  %4.1f %%", h);
        setLCD(0, String(line1)); setLCD(1, String(line2));
        logMsg("DHT22: T:" + String(t, 1) + "C, H:" + String(h, 1) + "%");
    } else {
        setLCD(0, "TEMP: ERROR");
        setLCD(1, "READ FAILED");
        logMsg("DHT22: Read error");
    }
    updateLCD();
}

void updateActivity() {
    lastActivityTime = millis();
    if (autoLockArmed && !isLocked) lastUserActivity = millis();
    if (!backlightOn) {
        lcd.backlight();
        backlightOn = true;
        lcdNeedsUpdate = true;
    }
}

void manageLCD() {
    if (millis() - lastActivityTime > SCREEN_TIMEOUT) {
        if (backlightOn) { lcd.noBacklight(); backlightOn = false; }
    } else {
        if (!backlightOn) { lcd.backlight(); backlightOn = true; lcdNeedsUpdate = true; }
    }

    if (currentDisplayMode == DISPLAY_TEMPERATURE) {
        if (millis() - displayStartTime > DISPLAY_DURATION) {
            logMsg("DISPLAY: Timeout - returning to main");
            showMainScreen();
        }
    }
    updateLCD();
}

void servoInit() {
    logMsg("SERVO: Allocating LEDC timers 0,1");

    logMsg("SERVO: Settled at " + String(target));
}

void connectWiFi() {
    logMsg("WIFI: Connecting to SSID: " + String(WIFI_SSID));
    WiFi.mode(WIFI_STA);
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 15000) {
        delay(500);
        Serial.print(".");
    }
    Serial.println();

    if (WiFi.status() == WL_CONNECTED) {
        online = true;
        logMsg("WIFI: Connected. IP=" + WiFi.localIP().toString());
    } else {
        online = false;
        logMsg("WIFI: Offline. Backdoor PIN 54321 still works.");
    }
}

bool ensureOnline() {
    online = (WiFi.status() == WL_CONNECTED);
    return online;
}

// Only touches /sensors, and only once per boot, and only after a
// successful round-trip proves Firebase is actually reachable.
void clearSensorsOnce() {
    if (sensorsClearedThisBoot) return;
    if (!ensureOnline()) return;
    if (firebaseDelete("/sensors")) {
        sensorsClearedThisBoot = true;
        logMsg("FIREBASE: /sensors cleared for fresh boot");
    }
}

// ============================================================
// PIN VALIDATION
// ------------------------------------------------------------
// Backdoor key always works offline. Every other PIN requires
// Firebase, so if we are offline the caller gets 'false' fast.
// ============================================================

bool validatePIN(String pin) {
    pin.trim();
    if (pin.length() == 0) return false;
    if (pin == BACKDOOR_KEY) return true;

    if (!ensureOnline()) return false;

    String raw = firebaseGet("/passkeys");
    if (raw.length() == 0) return false;

    String needle = "\"Passkey\":\"" + pin + "\"";
    return raw.indexOf(needle) >= 0;
}

// ============================================================
// CENTRAL STATE GATE
// ============================================================

bool preValidateAction(const String &action, const String &source) {
    String src = source; src.trim(); src.toUpperCase();

    if (action == "LOCK" && isLocked) {
        logMsg("PRE-VALIDATE: Reject LOCK - already locked (" + src + ")");
        showMessage("ALREADY LOCKED", ""); buzzerError(); delay(800); showMainScreen();
        if (src == "SERIAL") Serial.println("ERROR: Already locked");
        return false;
    }
    if (action == "UNLOCK" && !isLocked) {
        logMsg("PRE-VALIDATE: Reject UNLOCK - already unlocked (" + src + ")");
        showMessage("ALREADY UNLOCKED", ""); buzzerError(); delay(800); showMainScreen();
        if (src == "SERIAL") Serial.println("ERROR: Already unlocked");
        return false;
    }
    return true;
}

// ============================================================
// GATEKEEPER
// ============================================================

void handleServoAction(String action, String pin, String source) {
    action.trim(); action.toUpperCase();
    source.trim(); source.toUpperCase();

    logMsg("GATEKEEPER: action=" + action + " source=" + source +
           " state=" + getDoorStatus());

    if (!validatePIN(pin)) {
        logMsg("PIN: REJECTED (" + action + " from " + source + ")");
        showMessage("INVALID PIN", ""); buzzerError(); delay(1000); showMainScreen();
        if (source == "SERIAL") Serial.println("ERROR: Invalid PIN");
        return;
    }

    logMsg("PIN: ACCEPTED");

    if (action == "LOCK") {
        showActionMessage("LOCKING...");
        moveServo(SERVO_LOCK_POS);
        isLocked = true;
        disarmAutoLock();
        buzzerBeep(100, 1200); delay(500);
        showMainScreen();
        logMsg("SERVO: Door LOCKED");
        if (source == "SERIAL") Serial.println("SUCCESS: Locked");
        return;
    }
    if (action == "UNLOCK") {
        showActionMessage("UNLOCKING...");
        moveServo(SERVO_UNLOCK_POS);
        isLocked = false;
        unlockStartTime = millis();
        armAutoLock();
        buzzerSuccess(); delay(500);
        showMainScreen();
        logMsg("SERVO: Door UNLOCKED");
        if (source == "SERIAL") Serial.println("SUCCESS: Unlocked");
        return;
    }
    logMsg("SERVO: Unknown action - " + action);
    showMessage("UNKNOWN CMD", ""); buzzerError(); delay(800); showMainScreen();
}

// ============================================================
// OUTER VALIDATION
// ============================================================

void requestLockAction(String pin, String source) {
    source.trim(); source.toUpperCase();
    logMsg("REQUEST: LOCK from " + source + " (state=" + getDoorStatus() + ")");
    if (!preValidateAction("LOCK", source)) return;
    handleServoAction("LOCK", pin, source);
}

void requestUnlockAction(String pin, String source) {
    source.trim(); source.toUpperCase();
    logMsg("REQUEST: UNLOCK from " + source + " (state=" + getDoorStatus() + ")");
    if (!preValidateAction("UNLOCK", source)) return;
    handleServoAction("UNLOCK", pin, source);
}

// ------------------------------------------------------------
// Save a new passkey
// ------------------------------------------------------------
void savePasskeyToFirebase(const String &keyType, const String &passkey) {
    if (!ensureOnline()) {
        showMessage("OFFLINE", "Cannot save");
        buzzerError(); delay(1500);
        cancelEnrollment("offline");
        return;
    }

    int nextUID = 1;
    String nextStr = firebaseGet("/meta/next_uid");
    if (nextStr.length() > 0) nextUID = nextStr.toInt();
    if (nextUID < 1) nextUID = 1;

    String uid = String(nextUID);
    while (uid.length() < 3) uid = "0" + uid;

    String json = "{\"UID\":\"" + uid +
                  "\",\"Passkey\":\"" + passkey +
                  "\",\"Key_Type\":\"" + keyType + "\"}";

    bool ok = firebasePut("/passkeys/" + uid, json);
    if (ok) {
        firebasePut("/meta/next_uid", String(nextUID + 1));
        logMsg("FIREBASE: Saved /passkeys/" + uid + " type=" + keyType);
        showMessage("KEY SAVED", "UID: " + uid);
        buzzerSuccess();
    } else {
        logMsg("FIREBASE: Save failed");
        showMessage("SAVE FAILED", "");
        buzzerError();
    }
    delay(1500);
    enrollMode = ENROLL_NONE;
    enrollKeyType = "";
    enrollBuffer = "";
    showMainScreen();
}

// ------------------------------------------------------------
// Delete a passkey
// ------------------------------------------------------------
void deletePasskeyFromFirebase(const String &uid) {
    if (!ensureOnline()) {
        showMessage("OFFLINE", "");
        buzzerError(); delay(1500);
        cancelEnrollment("offline");
        return;
    }

    String existing = firebaseGet("/passkeys/" + uid + "/Passkey");
    if (existing.length() == 0) {
        logMsg("FIREBASE: UID " + uid + " not found");
        showMessage("NOT FOUND", "UID: " + uid);
        buzzerError(); delay(1500);
        enrollMode = ENROLL_NONE; enrollBuffer = "";
        showMainScreen();
        return;
    }

    if (firebaseDelete("/passkeys/" + uid)) {
        logMsg("FIREBASE: Deleted /passkeys/" + uid);
        showMessage("KEY REMOVED", "UID: " + uid);
        buzzerSuccess();
    } else {
        logMsg("FIREBASE: Delete failed");
        showMessage("DELETE FAILED", "");
        buzzerError();
    }
    delay(1500);
    enrollMode = ENROLL_NONE;
    enrollBuffer = "";
    showMainScreen();
}

// ------------------------------------------------------------
// Keypad handler while in enrollment
// ------------------------------------------------------------
void processEnrollKeypad(char key) {
    updateActivity();

    if (millis() - enrollStartTime > 30000) {
        cancelEnrollment("timeout");
        return;
    }

    if (enrollMode == ENROLL_SELECT_TYPE) {
        if (key == 'A' || key == 'a') {
            enrollKeyType = "PIN"; enrollMode = ENROLL_ENTER_PASSKEY;
            enrollBuffer = ""; enrollStartTime = millis();
            buzzerClick(); showEnrollPasskeyPrompt();
        } else if (key == 'B' || key == 'b') {
            enrollKeyType = "Key"; enrollMode = ENROLL_ENTER_PASSKEY;
            enrollBuffer = ""; enrollStartTime = millis();
            buzzerClick(); showEnrollPasskeyPrompt();
        } else if (key == 'C' || key == 'c') {
            enrollKeyType = "Card"; enrollMode = ENROLL_ENTER_PASSKEY;
            enrollBuffer = ""; enrollStartTime = millis();
            buzzerClick(); showEnrollPasskeyPrompt();
        } else if (key == 'D' || key == 'd') {
            cancelEnrollment("user cancelled");
        } else {
            buzzerClick();
        }
        return;
    }

    if (enrollMode == ENROLL_ENTER_PASSKEY) {
        if (key >= '0' && key <= '9') {
            if (enrollBuffer.length() < 16) {
                enrollBuffer += key; buzzerClick(); showEnrollPasskeyPrompt();
            }
        } else if ((enrollKeyType == "Key" || enrollKeyType == "Card") &&
                   (key == 'A' || key == 'a' || key == 'B' || key == 'b')) {
            if (enrollBuffer.length() < 16) {
                enrollBuffer += (char)toupper(key); buzzerClick(); showEnrollPasskeyPrompt();
            }
        } else if (key == 'C' || key == 'c') {
            if (enrollBuffer.length() == 0) { buzzerError(); return; }
            buzzerClick(); savePasskeyToFirebase(enrollKeyType, enrollBuffer);
        } else if (key == 'D' || key == 'd') {
            if (enrollBuffer.length() > 0) {
                enrollBuffer.remove(enrollBuffer.length() - 1);
                buzzerClick(); showEnrollPasskeyPrompt();
            }
        } else if (key == '*') {
            cancelEnrollment("user cancelled");
        }
        return;
    }

    if (enrollMode == ENROLL_ENTER_UID_REMOVE) {
        if (key >= '0' && key <= '9') {
            if (enrollBuffer.length() < 3) {
                enrollBuffer += key; buzzerClick(); showRemoveUidPrompt();
            }
        } else if (key == 'C' || key == 'c') {
            if (enrollBuffer.length() == 0) { buzzerError(); return; }
            String uid = enrollBuffer;
            while (uid.length() < 3) uid = "0" + uid;
            buzzerClick(); deletePasskeyFromFirebase(uid);
        } else if (key == 'D' || key == 'd') {
            if (enrollBuffer.length() > 0) {
                enrollBuffer.remove(enrollBuffer.length() - 1);
                buzzerClick(); showRemoveUidPrompt();
            }
        } else if (key == '*') {
            cancelEnrollment("user cancelled");
        }
        return;
    }
}

// ============================================================
// KEYPAD
// ============================================================

void processKeypad(char key) {
    if (millis() - lastKeyPressTime < DEBOUNCE_DELAY) return;
    lastKeyPressTime = millis();
    updateActivity();

    Serial.println("Keypad pressed: " + String(key));
    keypadOperationInProgress = true;

    if (enrollMode != ENROLL_NONE) {
        processEnrollKeypad(key);
        keypadOperationInProgress = false;
        return;
    }

    if (isPinMode) {
        if (!pinTimeoutTriggered && millis() - pinStartTime > PIN_TIMEOUT) {
            pinTimeoutTriggered = true;
            Serial.println("PIN: Timeout!");
            showMessage("TIMEOUT", "");
            buzzerError(); delay(1000);
            resetPinMode();
            keypadOperationInProgress = false;
            return;
        }

        if (key >= '0' && key <= '9') {
            if (pinBuffer.length() < 10) {
                pinBuffer += key;
                Serial.println("Debug buffer: " + pinBuffer);
                buzzerClick(); showPinEntry();
            }
        } else if (key == 'D' || key == 'd') {
            if (pinBuffer.length() > 0) {
                pinBuffer.remove(pinBuffer.length() - 1);
                Serial.println("Debug buffer: " + pinBuffer + " (deleted)");
                buzzerClick(); showPinEntry();
            }
        } else if (key == 'C' || key == 'c') {
            buzzerClick();
            String enteredPin = pinBuffer;
            String actionToRun = pendingAction;

            Serial.println("Debug buffer: " + enteredPin + " (confirmed)");
            Serial.println("Pending action: " + actionToRun);

            pinBuffer = "";
            isPinMode = false;
            pendingAction = "";
            pinStartTime = 0;
            pinTimeoutTriggered = false;

            if (actionToRun == "LOCK")        requestLockAction(enteredPin, "KEYPAD");
            else if (actionToRun == "UNLOCK") requestUnlockAction(enteredPin, "KEYPAD");
            else { Serial.println("PIN: No pending action - discarding"); showMainScreen(); }
        } else if (key == '*') {
            Serial.println("PIN: Cancelled by user");
            buzzerClick(); resetPinMode();
        } else {
            Serial.println("PIN: Ignoring key - " + String(key));
        }
        keypadOperationInProgress = false;
        return;
    }

    if (currentDisplayMode == DISPLAY_MAIN) {
        if (key == '#') {
            Serial.println("DISPLAY: Temperature");
            showTemperature(); buzzerClick();
        } else if (key == '*') {
            Serial.println("MAIN: Cancel (no-op)");
            buzzerClick();
        } else if (key == 'A' || key == 'a') {
            Serial.println("Keypad A: LOCK requested - enter PIN");
            pendingAction = "LOCK"; isPinMode = true; pinBuffer = "";
            pinStartTime = millis(); pinTimeoutTriggered = false;
            buzzerClick(); showPinEntry();
        } else if (key == 'B' || key == 'b') {
            Serial.println("Keypad B: UNLOCK requested - enter PIN");
            pendingAction = "UNLOCK"; isPinMode = true; pinBuffer = "";
            pinStartTime = millis(); pinTimeoutTriggered = false;
            buzzerClick(); showPinEntry();
        } else if (key >= '0' && key <= '9') {
            Serial.println("MAIN: Ignoring number - " + String(key));
        } else {
            Serial.println("MAIN: Ignoring key - " + String(key));
        }
    } else {
        Serial.println("DISPLAY: Ignoring key in non-main mode");
    }
    keypadOperationInProgress = false;
}

// ============================================================
// AUTO LOCK
// ============================================================

void armAutoLock() {
    autoLockArmed = true;
    lastUserActivity = millis();
    logMsg("AUTO-LOCK: Armed (" + String(AUTO_LOCK_DELAY / 1000) + "s idle)");
}

void disarmAutoLock() {
    if (autoLockArmed) logMsg("AUTO-LOCK: Disarmed");
    autoLockArmed = false;
}

void checkAutoLock() {
    if (!autoLockArmed) return;
    if (isLocked) { disarmAutoLock(); return; }
    if (isPinMode && (millis() - lastKeyPressTime < AUTO_LOCK_GRACE)) return;

    if (isPinMode && (millis() - lastKeyPressTime >= AUTO_LOCK_GRACE)) {
        logMsg("AUTO-LOCK: Cancelling stale PIN entry");
        pinBuffer = ""; isPinMode = false; pendingAction = "";
        pinStartTime = 0; pinTimeoutTriggered = false;
        showMainScreen();
    }

    if (currentDisplayMode != DISPLAY_MAIN) return;

    unsigned long idle = millis() - lastUserActivity;
    if (idle >= AUTO_LOCK_DELAY) {
        logMsg("AUTO-LOCK: Firing after " + String(idle / 1000) + "s idle");
        showMessage("AUTO-LOCKING", ""); buzzerBeep(100, 1000); delay(300);
        requestLockAction(BACKDOOR_KEY, "AUTO");
        disarmAutoLock();
        resetPinMode();
    }
}

// ============================================================
// DHT22
// ============================================================

bool initDHT() {
    logMsg("DHT22: Initializing...");
    dht.begin();
    delay(2000);
    for (int attempt = 0; attempt < 3; attempt++) {
        float h = dht.readHumidity();
        float t = dht.readTemperature();
        if (!isnan(h) && !isnan(t)) {
            lastTemperature = t; lastHumidity = h; hasDHTData = true;
            logMsg("DHT22: T:" + String(t, 1) + "C, H:" + String(h, 1) +
                   "% (attempt " + String(attempt + 1) + ")");
            return true;
        }
        logMsg("DHT22: Read attempt " + String(attempt + 1) + " failed");
        delay(2000);
    }
    logMsg("DHT22: All initialization attempts failed");
    hasDHTData = false;
    return false;
}

void updateDHTData() {
    if (millis() - lastDHTReadTime < DHT_READ_INTERVAL) return;
    lastDHTReadTime = millis();
    float h = dht.readHumidity();
    float t = dht.readTemperature();
    if (!isnan(h) && !isnan(t)) {
        lastTemperature = t; lastHumidity = h; hasDHTData = true;
    }
}

// ============================================================
// FIREBASE COMMAND LISTENER
// ============================================================

void handleFirebaseCommand(const String &rawCmd) {
    String cmd = rawCmd; cmd.trim(); cmd.toLowerCase();
    logMsg("FIREBASE CMD: " + cmd);

    if (cmd == "add_key") {
        startAddKeyFlow();
    } else if (cmd == "remove_key") {
        startRemoveKeyFlow();
    } else if (cmd == "lock") {
        requestLockAction(BACKDOOR_KEY, "FIREBASE");
    } else if (cmd == "unlock") {
        requestUnlockAction(BACKDOOR_KEY, "FIREBASE");
    } else {
        logMsg("FIREBASE CMD: Unknown command - " + cmd);
    }
}

void processFirebaseCommands() {
    if (!ensureOnline()) return;
    if (millis() - lastFirebaseCmdPoll < FIREBASE_CMD_POLL_INTERVAL) return;
    lastFirebaseCmdPoll = millis();

    String cmd = firebaseGet("/commands/request");
    cmd.trim();
    if (cmd.length() == 0) return;

    firebaseDelete("/commands/request");

    if (enrollMode != ENROLL_NONE) {
        logMsg("FIREBASE CMD: Ignored (mid-enrollment) - " + cmd);
        return;
    }
    handleFirebaseCommand(cmd);
}

// ============================================================
// SENSOR PUBLISH
// ============================================================

void pushSensorData() {
    if (!ensureOnline()) return;
    if (millis() - lastSensorPush < SENSOR_PUSH_INTERVAL) return;
    lastSensorPush = millis();

    clearSensorsOnce();

    float t = hasDHTData ? lastTemperature : 0.0f;
    float h = hasDHTData ? lastHumidity    : 0.0f;

    firebasePut("/sensors/temperature", String(t, 1));
    firebasePut("/sensors/humidity",    String(h, 1));
    firebasePut("/sensors/status",
                isLocked ? "\"locked\"" : "\"unlocked\"");
}

// ============================================================
// SERIAL COMMANDS
// ============================================================

void processSerialCommands() {
    if (!Serial.available()) return;

    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd.length() == 0) return;

    logMsg("SERIAL: Command - " + cmd);
    updateActivity();

    int colon = cmd.indexOf(':');
    String action, pin;
    if (colon > 0) {
        action = cmd.substring(0, colon);
        pin = cmd.substring(colon + 1);
        action.trim(); pin.trim();
    } else {
        action = cmd;
        pin = "";
    }
    action.toLowerCase();

    if (action == "lock")        requestLockAction(pin, "SERIAL");
    else if (action == "unlock") requestUnlockAction(pin, "SERIAL");
    else if (action == "status") {
        if (!validatePIN(pin)) {
            Serial.println("ERROR: Invalid PIN");
            showMessage("INVALID PIN", ""); buzzerError(); delay(800); showMainScreen();
            return;
        }
        Serial.println("STATUS: " + getDoorStatus());
        Serial.println("SERVO: " + String(currentServoPos) +
                       " EXPECTED: " + String(getExpectedServoPosition()) +
                       " CHECK: " + (isServoAtExpectedPosition() ? "OK" : "MISMATCH"));
        reportDoorStatus("SERIAL");
    } else if (action == "reset") {
        if (!validatePIN(pin)) { Serial.println("ERROR: Invalid PIN"); return; }
        logMsg("SERIAL: Resetting system");
        showMessage("RESETTING...", ""); delay(500);
        ESP.restart();
    } else if (action == "help") {
        Serial.println("\n=== SMART LOCK COMMANDS (v2.12) ===");
        Serial.println("Format: action:pin   (e.g. unlock:54321)");
        Serial.println("Keypad:");
        Serial.println("  A       - Request LOCK  (enter PIN, confirm with C)");
        Serial.println("  B       - Request UNLOCK (enter PIN, confirm with C)");
        Serial.println("  D       - Delete digit");
        Serial.println("  *       - Cancel PIN entry");
        Serial.println("  #       - Show Temperature (DHT22)");
        Serial.println("Serial:");
        Serial.println("  lock:54321     - Lock door");
        Serial.println("  unlock:54321   - Unlock door");
        Serial.println("  status:54321   - Get door status");
        Serial.println("  reset:54321    - Reset system");
        Serial.println("  help           - This menu");
        Serial.println("Firebase /commands/request:");
        Serial.println("  add_key        - Enroll new key on keypad");
        Serial.println("  remove_key     - Remove key by UID on keypad");
        Serial.println("  lock / unlock  - Direct door control");
        Serial.println("====================================\n");
    } else {
        Serial.println("Unknown command. Type 'help'.");
    }
}

// ============================================================
// RESET REASON
// ============================================================

void logResetReason() {
    esp_reset_reason_t reason = esp_reset_reason();
    const char *reasonStr = "UNKNOWN";
    switch (reason) {
        case ESP_RST_POWERON:   reasonStr = "POWERON"; break;
        case ESP_RST_EXT:       reasonStr = "EXT_PIN"; break;
        case ESP_RST_SW:        reasonStr = "SOFTWARE"; break;
        case ESP_RST_PANIC:     reasonStr = "PANIC/CRASH"; break;
        case ESP_RST_INT_WDT:   reasonStr = "INT_WDT"; break;
        case ESP_RST_TASK_WDT:  reasonStr = "TASK_WDT"; break;
        case ESP_RST_WDT:       reasonStr = "WDT"; break;
        case ESP_RST_DEEPSLEEP: reasonStr = "DEEPSLEEP"; break;
        case ESP_RST_BROWNOUT:  reasonStr = "BROWNOUT"; break;
        case ESP_RST_SDIO:      reasonStr = "SDIO"; break;
        default:                reasonStr = "UNKNOWN"; break;
    }
    logMsg("RESET REASON: " + String(reasonStr) + " (" + String((int)reason) + ")");
}

void setup() {
    delay(1000);
    Serial.begin(115200);
    delay(100);
    systemStartTime = millis();

    logMsg("========================================");
    logMsg("ESP32 Smart Lock v2.12 Starting...");
    logMsg("========================================");
    logResetReason();

    logMsg("LCD: Initializing...");
    Wire.begin();
    Wire.setClock(100000);

    lcd.backlight();
    lcd.clear();
    backlightOn = true;
    showMessage("Smart Lock", "v2.12 Starting...");
    delay(1000);

    logMsg("BUZZER: Initializing...");
    buzzerInit();
    buzzerBeep(100, 1500); delay(100);
    buzzerBeep(100, 1500);

    logMsg("SERVO: Initializing...");
    showMessage("Smart Lock", "Servo...");
    servoInit();
    delay(300);

    showMessage("Smart Lock", "DHT22...");
    initDHT();

    showMessage("Smart Lock", "WiFi...");
    connectWiFi();

    delay(500);
    showMainScreen();

    logMsg("========================================");
    logMsg(online ? "SYSTEM: Online (WiFi + Firebase)"
                  : "SYSTEM: Offline (backdoor PIN only)");
    logMsg("KEYPAD: A=Lock B=Unlock  #=Temperature");
    logMsg("SERIAL: lock:pin / unlock:pin / status:pin / reset:pin / help");
    logMsg("========================================");

    buzzerSuccess();
}

// ============================================================
// LOOP
// ------------------------------------------------------------
// Keypad is polled FIRST so a slow HTTP call can never eat a
// press. Everything else is only allowed to run after we know
// the user input has been consumed.
// ============================================================

void loop() {
    // 1. Input first — never starve the keypad.
    char key = keypad.getKey();
    if (key) processKeypad(key);

    processSerialCommands();

    // 2. UI update.
    checkAutoLock();
    manageLCD();

    // 3. Sensor sampling (local, no network).
    updateDHTData();

    // 4. Network work — all bounded by HTTP_TIMEOUT_MS.
    processFirebaseCommands();
    pushSensorData();

    static unsigned long lastCountdownLog = 0;
    if (autoLockArmed && !isLocked && millis() - lastCountdownLog > 5000) {
        lastCountdownLog = millis();
        unsigned long idle = millis() - lastUserActivity;
        long remaining = (long)AUTO_LOCK_DELAY - (long)idle;
        if (remaining < 0) remaining = 0;
        Serial.println("[AUTO-LOCK] Idle " + String(idle / 1000) +
                       "s, firing in " + String(remaining / 1000) + "s");
    }

    delay(2);
}
