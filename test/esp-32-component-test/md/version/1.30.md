String getTimestamp()
{
    unsigned long elapsed = millis() - systemStartTime;
    unsigned long seconds = elapsed / 1000;
    unsigned long minutes = seconds / 60;
    unsigned long hours = minutes / 60;
    seconds %= 60;
    minutes %= 60;
    char buf[20];
    snprintf(buf, sizeof(buf), "%02lu:%02lu:%02lu", hours, minutes, seconds);
    return String(buf);
}

void logMsg(String msg)
{
    Serial.println("[" + getTimestamp() + "] " + msg);
}

void setLCD(int row, const String &text)
{
    String trimmed = text;
    if (trimmed.length() > 16)
        trimmed = trimmed.substring(0, 16);
    while (trimmed.length() < 16)
        trimmed += " ";
    if (row == 0)
    {
        if (lcdRow0 != trimmed)
        {
            lcdRow0 = trimmed;
            lcdNeedsUpdate = true;
        }
    }
    else
    {
        if (lcdRow1 != trimmed)
        {
            lcdRow1 = trimmed;
            lcdNeedsUpdate = true;
        }
    }
}

void updateLCD()
{
    if (!lcdNeedsUpdate)
        return;
    if (millis() - lastLCDUpdateTime < LCD_UPDATE_INTERVAL)
        return;
    lastLCDUpdateTime = millis();
    lcd.setCursor(0, 0);
    lcd.print(lcdRow0);
    lcd.setCursor(0, 1);
    lcd.print(lcdRow1);
    lcdNeedsUpdate = false;
}

void showMainScreen()
{
    currentDisplayMode = DISPLAY_MAIN;
    String status = isLocked ? "LOCKED" : "UNLOCKED";
    setLCD(0, status);
    setLCD(1, "A=LOCK B=UNLOCK");
    updateLCD();
}

void showPinEntry()

    setLCD(0, "ENTER PIN");

    if (MASK_PIN_ON_LCD)

            display += "*";
    }
    else

        display = pinBuffer;

    while (display.length() < 16)
        display += " ";
    setLCD(1, display);
    updateLCD();
}

}

void showTemperature()
{
    currentDisplayMode = DISPLAY_TEMPERATURE;
    displayStartTime = millis();
    float h = dht.readHumidity();
    float t = dht.readTemperature();
    if (!isnan(h) && !isnan(t))
    {
        lastTemperature = t;
        lastHumidity = h;
        hasDHTData = true;
        char l1[17], l2[17];
        snprintf(l1, sizeof(l1), "TEMP: %2.0f C", t);
        snprintf(l2, sizeof(l2), "HUM:  %2.0f %%", h);
        setLCD(0, String(l1));
        setLCD(1, String(l2));
        logMsg("DHT: T:" + String(t, 1) + "C, H:" + String(h, 1) + "%");
    }
    else
    {
        delay(100);
        h = dht.readHumidity();
        t = dht.readTemperature();
        if (!isnan(h) && !isnan(t))
        {
            lastTemperature = t;
            lastHumidity = h;
            hasDHTData = true;
            char l1[17], l2[17];
            snprintf(l1, sizeof(l1), "TEMP: %2.0f C", t);
            snprintf(l2, sizeof(l2), "HUM:  %2.0f %%", h);
            setLCD(0, String(l1));
            setLCD(1, String(l2));
        }
        else
        {
            setLCD(0, "TEMP: ERROR");
            setLCD(1, "READ FAILED");
        }
    }
    updateLCD();
}

void updateActivity()
{
    lastActivityTime = millis();
    if (autoLockArmed && !isLocked)
        lastUserActivity = millis();
    if (!lcd.getBacklight())
    {
        lcd.backlight();
        lcdNeedsUpdate = true;
    }
}

void manageLCD()
{
    if (millis() - lastActivityTime > SCREEN_TIMEOUT)
    {
        if (lcd.getBacklight())
            lcd.noBacklight();
    }
    else
    {
        if (!lcd.getBacklight())
        {
            lcd.backlight();
            lcdNeedsUpdate = true;
        }
    }

    if (currentDisplayMode == DISPLAY_TEMPERATURE ||
        currentDisplayMode == DISPLAY_BLUETOOTH)
    {
        if (millis() - displayStartTime > DISPLAY_DURATION)
        {
            showMainScreen();
        }
    }
    updateLCD();
}

bool firebasePut(const String &path, const String &value)
{
    if (WiFi.status() != WL_CONNECTED)
        return false;
    HTTPClient http;
    String url = String(FIREBASE_URL) + "/" + path + ".json?auth=" + FIREBASE_AUTH;
    http.begin(url);
    http.setTimeout(HTTP_TIMEOUT_MS);
    http.addHeader("Content-Type", "application/json");
    int code = http.PUT(value);
    http.end();
    if (code < 200 || code >= 300)
    {
        logMsg("FB PUT " + path + " failed (HTTP " + String(code) + ")");
        return false;
    }
    return true;
}

String firebaseGet(const String &path)
{
    if (WiFi.status() != WL_CONNECTED)
        return "";
    HTTPClient http;
    String url = String(FIREBASE_URL) + "/" + path + ".json?auth=" + FIREBASE_AUTH;
    http.begin(url);
    http.setTimeout(HTTP_TIMEOUT_MS);
    int code = http.GET();
    String payload = "";
    if (code > 0)
        payload = http.getString();
    http.end();
    if (code < 200 || code >= 300)
    {
        logMsg("FB GET " + path + " failed (HTTP " + String(code) + ")");
        return "";
    }
    payload.trim();
    return payload;
}

// ============================================================
// FIREBASE — PASSKEY LOOKUP / VALIDATION
// ============================================================
String findPasskeyUID(const String &value, const String &keyType)
{
    if (WiFi.status() != WL_CONNECTED)
        return "";
    String body = firebaseGet("passkeys");
    if (body.length() < 3 || body == "null")
        return "";

    JsonDocument doc;
    DeserializationError err = deserializeJson(doc, body);
    if (err)
    {
        logMsg("FB passkeys parse error: " + String(err.c_str()));
        return "";
    }

    for (JsonPair kv : doc.as<JsonObject>())
    {
        const char *uid = kv.key().c_str();
        String inner = kv.value().as<String>();

        JsonDocument rec;
        DeserializationError e2 = deserializeJson(rec, inner);
        if (e2)
            continue;

        String storedPass = rec["Passkey"].as<String>();
        String storedType = rec["Key_Type"].as<String>();

        if (storedPass == value && storedType.equalsIgnoreCase(keyType))
        {
            logMsg("FB passkey match: UID=" + String(uid) + " type=" + storedType);
            return String(uid);
        }
    }
    return "";
}

String nextAvailableUID()
{
    if (WiFi.status() != WL_CONNECTED)
        return "001";
    String body = firebaseGet("passkeys");
    if (body.length() < 3 || body == "null")
        return "001";

    JsonDocument doc;
    DeserializationError err = deserializeJson(doc, body);
    if (err)
        return "001";

    int maxId = 0;
    for (JsonPair kv : doc.as<JsonObject>())
    {
        int v = atoi(kv.key().c_str());
        if (v > maxId)
            maxId = v;
    }
    char buf[8];
    snprintf(buf, sizeof(buf), "%03d", maxId + 1);
    return String(buf);
}

// ============================================================
// PIN VALIDATION
// ============================================================
bool validatePIN(const String &pin)
{
    String p = pin;
    p.trim();
    if (p.length() == 0)
        return false;

    if (WiFi.status() == WL_CONNECTED)
    {
        String uid = findPasskeyUID(p, "PIN");
        if (uid.length() > 0)
            return true;
    }

    if (p == BACKDOOR_KEY)
        return true;

    return false;
}

// ============================================================
// RFID VALIDATION
// ============================================================
bool validateRFID(const String &uid)
{
    if (uid.length() == 0)
        return false;

    if (WiFi.status() == WL_CONNECTED)
    {
        String match = findPasskeyUID(uid, "Card");
        if (match.length() > 0)
            return true;
    }

    return false;
}

// ============================================================
// CENTRAL STATE GATE
// ============================================================
bool preValidateAction(const String &action, const String &source)
{
    String src = source;
    src.trim();
    src.toUpperCase();

    if (action == "LOCK" && isLocked)
    {
        logMsg("PRE-VALIDATE: reject LOCK - already locked (src=" + src + ")");
        showMessage("ALREADY LOCKED", "");
        buzzerError();
        delay(800);
        showMainScreen();
        if (src == "BLUETOOTH")
            sendResponse("ERROR: Already locked");
        else if (src == "SERIAL")
            Serial.println("ERROR: Already locked");
        return false;
    }
    if (action == "UNLOCK" && !isLocked)
    {
        logMsg("PRE-VALIDATE: reject UNLOCK - already unlocked (src=" + src + ")");
        showMessage("ALREADY UNLOCKED", "");
        buzzerError();
        delay(800);
        showMainScreen();
        if (src == "BLUETOOTH")
            sendResponse("ERROR: Already unlocked");
        else if (src == "SERIAL")
            Serial.println("ERROR: Already unlocked");
        return false;
    }
    return true;
}

// ============================================================
// LAYER 2: GATEKEEPER — PIN CHECK + DISPATCH
// ============================================================
void handleServoAction(String action, String pin, String source)
{
    action.trim();
    action.toUpperCase();
    source.trim();
    source.toUpperCase();

    logMsg("GATEKEEPER: action=" + action + " src=" + source +
           " state=" + getDoorStatus());

    if (!validatePIN(pin))
    {
        logMsg("PIN: REJECTED (src=" + source + ")");
        showMessage("INVALID PIN", "");
        buzzerError();
        delay(1000);
        showMainScreen();
        if (source == "BLUETOOTH")
            sendResponse("ERROR: Invalid PIN");
        else if (source == "SERIAL")
            Serial.println("ERROR: Invalid PIN");
        return;
    }
    logMsg("PIN: ACCEPTED");

    if (action == "LOCK")
    {
        showMessage("LOCKING...", "");
        moveServo(SERVO_LOCK_POS);
        isLocked = true;
        disarmAutoLock();
        buzzerBeep(100, 1200);
        delay(500);
        showMainScreen();
        logMsg("SERVO: Door LOCKED");
        publishStatus();
        if (source == "BLUETOOTH")
            sendResponse("SUCCESS: Locked");
        else if (source == "SERIAL")
            Serial.println("SUCCESS: Locked");
        return;
    }

    if (action == "UNLOCK")
    {
        showMessage("UNLOCKING...", "");
        moveServo(SERVO_UNLOCK_POS);
        isLocked = false;
        unlockStartTime = millis();
        armAutoLock();
        buzzerSuccess();
        delay(500);
        showMainScreen();
        logMsg("SERVO: Door UNLOCKED");
        publishStatus();
        if (source == "BLUETOOTH")
            sendResponse("SUCCESS: Unlocked");
        else if (source == "SERIAL")
            Serial.println("SUCCESS: Unlocked");
        return;
    }

    logMsg("SERVO: Unknown action - " + action);
    showMessage("UNKNOWN CMD", "");
    buzzerError();
    delay(800);
    showMainScreen();
}

// ============================================================
// LAYER 1: OUTER VALIDATION — STATE CHECK + DISPATCH
// ============================================================
void requestLockAction(String pin, String source)
{
    source.trim();
    source.toUpperCase();
    logMsg("REQUEST: LOCK from " + source + " (state=" + getDoorStatus() + ")");
    if (!preValidateAction("LOCK", source))
        return;
    handleServoAction("LOCK", pin, source);
}
void requestUnlockAction(String pin, String source)
{
    source.trim();
    source.toUpperCase();
    logMsg("REQUEST: UNLOCK from " + source + " (state=" + getDoorStatus() + ")");
    if (!preValidateAction("UNLOCK", source))
        return;
    handleServoAction("UNLOCK", pin, source);
}

bool processFirebaseSessionKey(char key)
{
    if (!firebaseSessionActive)
        return false;

    lastSessionActivity = millis();
    updateActivity();

    if (currentDisplayMode == DISPLAY_ADD_KEY_SELECT)
    {
        if (key == 'A' || key == 'a')
        {
            currentDisplayMode = DISPLAY_ADD_KEY_PIN;
            sessionBuffer = "";
            setLCD(0, "NEW PIN (>=4 dig)");
            setLCD(1, "C=OK D=DEL * =X");
            updateLCD();
            buzzerClick();
            return true;
        }
        if (key == 'B' || key == 'b')
        {
            setLCD(0, "KEY TYPE: USE APP");
            setLCD(1, "No reader onboard");
            updateLCD();
            buzzerError();
            delay(1500);
            finishFirebaseSession();
            return true;
        }
        if (key == 'C' || key == 'c')
        {
            currentDisplayMode = DISPLAY_ADD_KEY_CARD;
            sessionBuffer = "";
            setLCD(0, "SCAN RFID CARD");
            setLCD(1, "* =CANCEL");
            updateLCD();
            buzzerClick();
            return true;
        }
        if (key == 'D' || key == 'd' || key == '*')
        {
            logMsg("SESSION: user cancelled select");
            buzzerClick();
            finishFirebaseSession();
            return true;
        }
        return true;
    }

    if (currentDisplayMode == DISPLAY_ADD_KEY_PIN)
    {
        if (key >= '0' && key <= '9')
        {
            if (sessionBuffer.length() < 10)
            {
                sessionBuffer += key;
                String d = sessionBuffer;
                while (d.length() < 16)
                    d += " ";
                setLCD(1, d);
                updateLCD();
                buzzerClick();
            }
            return true;
        }
        if (key == 'D' || key == 'd')
        {
            if (sessionBuffer.length() > 0)
            {
                sessionBuffer.remove(sessionBuffer.length() - 1);
                String d = sessionBuffer;
                while (d.length() < 16)
                    d += " ";
                setLCD(1, d);
                updateLCD();
                buzzerClick();
            }
            return true;
        }
        if (key == 'C' || key == 'c')
        {
            if (sessionBuffer.length() < 4)
            {
                setLCD(0, "PIN TOO SHORT");
                updateLCD();
                buzzerError();
                delay(800);
                finishFirebaseSession();
                return true;
            }
            String newUID = nextAvailableUID();
            String record = "{\"UID\":\"" + newUID +
                            "\",\"Passkey\":\"" + sessionBuffer +
                            "\",\"Key_Type\":\"PIN\"}";
            String payload = "\"";
            for (unsigned int i = 0; i < record.length(); i++)
            {
                char c = record[i];
                if (c == '"')
                    payload += "\\\"";
                else
                    payload += c;
            }
            payload += "\"";

            bool ok = firebasePut("passkeys/" + newUID, payload);
            if (ok)
            {
                logMsg("SESSION: added PIN UID=" + newUID);
                setLCD(0, "ADDED UID " + newUID);
                setLCD(1, "");
                updateLCD();
                buzzerSuccess();
                delay(1500);
            }
            else
            {
                setLCD(0, "ADD FAILED");
                setLCD(1, "Check WiFi");
                updateLCD();
                buzzerError();
                delay(1500);
            }
            finishFirebaseSession();
            return true;
        }
        if (key == '*')
        {
            buzzerClick();
            finishFirebaseSession();
            return true;
        }
        return true;
    }

    if (currentDisplayMode == DISPLAY_ADD_KEY_CARD)
    {
        if (key == '*')
        {
            buzzerClick();
            finishFirebaseSession();
            return true;
        }
        return true;
    }

    if (currentDisplayMode == DISPLAY_REMOVE_KEY_UID)
    {
        if (key >= '0' && key <= '9')
        {
            if (sessionBuffer.length() < 3)
            {
                sessionBuffer += key;
                String d = "UID: " + sessionBuffer;
                while (d.length() < 16)
                    d += " ";
                setLCD(1, d);
                updateLCD();
                buzzerClick();
            }
            return true;
        }
        if (key == 'D' || key == 'd')
        {
            if (sessionBuffer.length() > 0)
            {
                sessionBuffer.remove(sessionBuffer.length() - 1);
                String d = "UID: " + sessionBuffer;
                while (d.length() < 16)
                    d += " ";
                setLCD(1, d);
                updateLCD();
                buzzerClick();
            }
            return true;
        }
        if (key == 'C' || key == 'c')
        {
            if (sessionBuffer.length() == 0)
            {
                buzzerError();
                return true;
            }
            String uid = sessionBuffer;
            while (uid.length() < 3)
                uid = "0" + uid;

            String existing = firebaseGet("passkeys/" + uid);
            if (existing.length() > 0 && existing != "null")
            {
                bool ok = firebaseDelete("passkeys/" + uid);
                if (ok)
                {
                    logMsg("SESSION: removed UID=" + uid);
                    setLCD(0, "REMOVED UID " + uid);
                    setLCD(1, "");
                    updateLCD();
                    buzzerSuccess();
                }
                else
                {
                    setLCD(0, "REMOVE FAILED");
                    setLCD(1, "Check WiFi");
                    updateLCD();
                    buzzerError();
                }
            }
            else
            {
                setLCD(0, "UID NOT FOUND");
                setLCD(1, "UID " + uid);
                updateLCD();
                buzzerError();
            }
            delay(1500);
            finishFirebaseSession();
            return true;
        }
        if (key == '*')
        {
            buzzerClick();
            finishFirebaseSession();
            return true;
        }
        return true;
    }

    return false;
}

void handleFirebaseCommand(const String &cmdRaw)
{
    String cmd = cmdRaw;
    cmd.trim();
    cmd.toLowerCase();
    logMsg("FIREBASE: command = '" + cmd + "'");

    if (cmd == "add_key")
    {
        startAddKeySession();
    }
    else if (cmd == "remove_key")
    {
        startRemoveKeySession();
    }
    else if (cmd == "lock")
    {
        requestLockAction(BACKDOOR_KEY, "FIREBASE");
    }
    else if (cmd == "unlock")
    {
        requestUnlockAction(BACKDOOR_KEY, "FIREBASE");
    }
    else if (cmd.length() == 0)
    {
    }
    else
    {
        logMsg("FIREBASE: unknown command ignored");
    }
}

void pollFirebaseCommands()
{
    if (WiFi.status() != WL_CONNECTED)
        return;
    if (firebaseSessionActive)
        return;

    String raw = firebaseGet("commands/request");
    if (raw.length() == 0)
        return;

    String cmd = stripOuterQuotes(raw);
    if (cmd.length() == 0 || cmd == "null")
        return;

    firebasePut("commands/request", "\"\"");
    handleFirebaseCommand(cmd);
}

        return;

    if (hasDHTData)
    {

        {
            if (firebasePut("sensors/temperature", t))
            {
                lastPushedTemp = t;
            }
        }
        String h = String(lastHumidity, 1);
        if (h != lastPushedHum)
        {
            if (firebasePut("sensors/humidity", h))
            {
                lastPushedHum = h;
            }
        }
    }
}

void ensureWiFi()
{
    if (WiFi.status() == WL_CONNECTED)
    {
        if (!wifiConnected)
        {
            wifiConnected = true;
            logMsg("WIFI: connected, IP=" + WiFi.localIP().toString());
            lastPushedStatus = "";
            lastPushedTemp = "";
            lastPushedHum = "";
        }
        return;
    }

    if (wifiConnected)
    {
        logMsg("WIFI: disconnected");
        wifiConnected = false;
    }

    logMsg("WIFI: connecting to " + String(WIFI_SSID) + " ...");
    WiFi.mode(WIFI_STA);
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);

    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 8000)
    {
        delay(200);
    }
    if (WiFi.status() == WL_CONNECTED)
    {
        wifiConnected = true;
        logMsg("WIFI: connected, IP=" + WiFi.localIP().toString());
        lastPushedStatus = "";
        lastPushedTemp = "";
        lastPushedHum = "";
    }
    else
    {
        logMsg("WIFI: connect failed (will retry)");
    }
}

// KEYPAD
// ============================================================
void processKeypad(char key)
{
    if (millis() - lastKeyPressTime < DEBOUNCE_DELAY)
        return;
    lastKeyPressTime = millis();
    updateActivity();
    logMsg("KEY: " + String(key));
    keypadOperationInProgress = true;

    if (firebaseSessionActive)
    {
        processFirebaseSessionKey(key);
        keypadOperationInProgress = false;
        return;
    }

    if (isPinMode)
    {
        if (!pinTimeoutTriggered && millis() - pinStartTime > PIN_TIMEOUT)
        {
            pinTimeoutTriggered = true;
            showMessage("TIMEOUT", "");
            buzzerError();
            delay(1000);
            resetPinMode();
            keypadOperationInProgress = false;
            return;
        }

        if (key >= '0' && key <= '9')
        {
            if (pinBuffer.length() < 10)
            {
                pinBuffer += key;
                buzzerClick();
                showPinEntry();
            }
        }
        else if (key == 'D' || key == 'd')
        {
            if (pinBuffer.length() > 0)
            {
                pinBuffer.remove(pinBuffer.length() - 1);
                buzzerClick();
                showPinEntry();
            }
        }
        else if (key == 'C' || key == 'c')
        {
            buzzerClick();
            String enteredPin = pinBuffer;
            String actionToRun = pendingAction;
            pinBuffer = "";
            isPinMode = false;
            pendingAction = "";
            pinStartTime = 0;
            pinTimeoutTriggered = false;

            if (actionToRun == "LOCK")
                requestLockAction(enteredPin, "KEYPAD");
            else if (actionToRun == "UNLOCK")
                requestUnlockAction(enteredPin, "KEYPAD");
            else
                showMainScreen();
        }
        else if (key == '*')
        {
            buzzerClick();
            resetPinMode();
        }
        keypadOperationInProgress = false;
        return;
    }

    if (currentDisplayMode == DISPLAY_MAIN)
    {
        if (key == '#')
        {
            showTemperature();
            buzzerClick();
        }
        else if (key == '*')
        {
            showBluetoothInfo();
            buzzerClick();
        }
        else if (key == 'A' || key == 'a')
        {
            pendingAction = "LOCK";
            isPinMode = true;
            pinBuffer = "";
            pinStartTime = millis();
            pinTimeoutTriggered = false;
            buzzerClick();
            showPinEntry();
        }
        else if (key == 'B' || key == 'b')
        {
            pendingAction = "UNLOCK";
            isPinMode = true;
            pinBuffer = "";
            pinStartTime = millis();
            pinTimeoutTriggered = false;
            buzzerClick();
            showPinEntry();
        }
    }

    keypadOperationInProgress = false;
}

// ============================================================
// RFID
// ============================================================
void processRFID()
{
    if (!rfidInitialized)
        return;
    if (millis() - rfidCooldown < RFID_COOLDOWN)
        return;

    static unsigned long lastCheck = 0;
    if (millis() - lastCheck < 30)
        return;
    lastCheck = millis();

    if (!mfrc522.PICC_IsNewCardPresent())
        return;
    if (!mfrc522.PICC_ReadCardSerial())
        return;

    String uid = "";
    for (byte i = 0; i < mfrc522.uid.size; i++)
    {
        if (mfrc522.uid.uidByte[i] < 0x10)
            uid += "0";
        uid += String(mfrc522.uid.uidByte[i], HEX);
    }
    uid.toUpperCase();
    logMsg("RFID: UID=" + uid);

    updateActivity();

    if (firebaseSessionActive && currentDisplayMode == DISPLAY_ADD_KEY_CARD)
    {
        String newUID = nextAvailableUID();
        String record = "{\"UID\":\"" + newUID +
                        "\",\"Passkey\":\"" + uid +
                        "\",\"Key_Type\":\"Card\"}";
        String payload = "\"";
        for (unsigned int i = 0; i < record.length(); i++)
        {
            char c = record[i];
            if (c == '"')
                payload += "\\\"";
            else
                payload += c;
        }
        payload += "\"";

        setLCD(0, "SAVING CARD...");
        setLCD(1, uid);
        updateLCD();

        bool ok = firebasePut("passkeys/" + newUID, payload);
        if (ok)
        {
            logMsg("SESSION: added CARD UID=" + newUID + " tag=" + uid);
            setLCD(0, "ADDED " + newUID);
            setLCD(1, "TYPE: CARD");
            updateLCD();
            buzzerSuccess();
        }
        else
        {
            setLCD(0, "ADD FAILED");
            setLCD(1, "Check WiFi");
            updateLCD();
            buzzerError();
        }
        delay(1500);
        finishFirebaseSession();

        mfrc522.PICC_HaltA();
        mfrc522.PCD_StopCrypto1();
        rfidCooldown = millis();
        return;
    }

    showMessage(uid, "SCANNED");
    buzzerClick();
    delay(300);

    if (isPinMode)
        resetPinMode();

    bool accepted = validateRFID(uid);

    if (!accepted && !ALLOW_UNKNOWN_TAGS)
    {
        logMsg("RFID: tag rejected (not in /passkeys)");
        showMessage("ACCESS DENIED", uid);
        buzzerError();
        delay(1200);
        showMainScreen();
    }
    else
    {
        if (isLocked)
        {
            handleServoAction("UNLOCK", BACKDOOR_KEY, "RFID");
        }
        else
        {
            handleServoAction("LOCK", BACKDOOR_KEY, "RFID");
        }
    }

    mfrc522.PICC_HaltA();
    mfrc522.PCD_StopCrypto1();
    rfidCooldown = millis();
}

bool parseBTCommand(String cmd, String &action, String &pin)
{
    cmd.trim();
    int colon = cmd.indexOf(':');
    if (colon > 0)
    {
        action = cmd.substring(0, colon);
        pin = cmd.substring(colon + 1);
        action.trim();
        pin.trim();
        action.toUpperCase();
        return true;
    }
    if (cmd.startsWith(BACKDOOR_KEY))
    {
        String rest = cmd.substring(BACKDOOR_KEY.length());
        rest.trim();
        rest.toUpperCase();
        if (rest == "_BT_LOCK")
        {
            action = "LOCK";
            pin = BACKDOOR_KEY;
            return true;
        }
        if (rest == "_BT_UNLOCK")
        {
            action = "UNLOCK";
            pin = BACKDOOR_KEY;
            return true;
        }
        if (rest == "_BT_STATUS")
        {
            action = "STATUS";
            pin = BACKDOOR_KEY;
            return true;
        }
    }
    return false;
}

void processBT(String cmd)
{
    cmd.trim();
    logMsg("BT: command - " + cmd);
    updateActivity();

    String action, pin;
    if (!parseBTCommand(cmd, action, pin))
    {
        sendResponse("ERROR: Invalid format. Use LOCK:pin / UNLOCK:pin / STATUS:pin");
        showMessage("INVALID CMD", "");
        buzzerError();
        delay(800);
        showMainScreen();
        return;
    }

    if (action == "STATUS")
    {
        if (!validatePIN(pin))
        {
            sendResponse("ERROR: Invalid PIN");
            showMessage("INVALID PIN", "");
            buzzerError();
            delay(800);
            showMainScreen();
            return;
        }
        String s = getDoorStatus();
        showMessage("STATUS", s);
        delay(800);
        showMainScreen();
        sendResponse("STATUS: " + s);
        sendResponse("SERVO: " + String(currentServoPos) +
                     " EXPECTED: " + String(getExpectedServoPosition()) +
                     " CHECK: " + (isServoAtExpectedPosition() ? "OK" : "MISMATCH"));
        reportDoorStatus("BT");
        return;
    }
    if (action == "LOCK")
    {
        requestLockAction(pin, "BLUETOOTH");
        return;
    }
    if (action == "UNLOCK")
    {
        requestUnlockAction(pin, "BLUETOOTH");
        return;
    }

    sendResponse("ERROR: Unknown action");
    showMessage("UNKNOWN CMD", "");
    buzzerError();
    delay(800);
    showMainScreen();
}

void armAutoLock()
{
    autoLockArmed = true;
    lastUserActivity = millis();
    logMsg("AUTO-LOCK: armed (" + String(AUTO_LOCK_DELAY / 1000) + "s idle)");
}
void disarmAutoLock()
{
    if (autoLockArmed)
        logMsg("AUTO-LOCK: disarmed");
    autoLockArmed = false;
}
void checkAutoLock()
{
    if (!autoLockArmed)
        return;
    if (isLocked)
    {
        disarmAutoLock();
        return;
    }

    if (isPinMode && (millis() - lastKeyPressTime < AUTO_LOCK_GRACE))
        return;

    if (isPinMode && (millis() - lastKeyPressTime >= AUTO_LOCK_GRACE))
    {
        logMsg("AUTO-LOCK: cancelling stale PIN entry");
        pinBuffer = "";
        isPinMode = false;
        pendingAction = "";
        pinStartTime = 0;
        pinTimeoutTriggered = false;
        showMainScreen();
    }

    if (firebaseSessionActive)
        return;
    if (currentDisplayMode != DISPLAY_MAIN)
        return;

    unsigned long idle = millis() - lastUserActivity;
    if (idle >= AUTO_LOCK_DELAY)
    {
        logMsg("AUTO-LOCK: firing after " + String(idle / 1000) + "s idle");
        showMessage("AUTO-LOCKING", "");
        buzzerBeep(100, 1000);
        delay(300);
        requestLockAction(BACKDOOR_KEY, "AUTO");
        disarmAutoLock();
        resetPinMode();
    }
}

bool initDHT()
{
    logMsg("DHT: init...");
    dht.begin();
    delay(1000);
    for (int attempt = 0; attempt < 3; attempt++)
    {
        float h = dht.readHumidity();
        float t = dht.readTemperature();
        if (!isnan(h) && !isnan(t))
        {
            lastTemperature = t;
            lastHumidity = h;
            hasDHTData = true;
            logMsg("DHT: T:" + String(t, 1) + "C H:" + String(h, 1) + "% (try " + String(attempt + 1) + ")");
            return true;
        }
        delay(500);
    }
    hasDHTData = false;
    logMsg("DHT: init failed");
    return false;
}

void updateDHTData()
{
    if (millis() - lastDHTReadTime < DHT_READ_INTERVAL)
        return;
    lastDHTReadTime = millis();
    float h = dht.readHumidity();
    float t = dht.readTemperature();
    if (!isnan(h) && !isnan(t))
    {
        lastTemperature = t;
        lastHumidity = h;
        hasDHTData = true;
    }
}

// SERIAL
// ============================================================
void processSerialCommands()
{
    if (!Serial.available())
        return;
    String cmd = Serial.readStringUntil('\n');
    cmd.trim();
    if (cmd.length() == 0)
        return;

    logMsg("SERIAL: " + cmd);
    updateActivity();

    int colon = cmd.indexOf(':');
    String action, pin;
    if (colon > 0)
    {
        action = cmd.substring(0, colon);
        pin = cmd.substring(colon + 1);
        action.trim();
        pin.trim();
    }
    else
    {
        action = cmd;
        pin = "";
    }
    action.toLowerCase();

    if (action == "lock")
        requestLockAction(pin, "SERIAL");
    else if (action == "unlock")
        requestUnlockAction(pin, "SERIAL");
    else if (action == "status")
    {
        if (!validatePIN(pin))
        {
            Serial.println("ERROR: Invalid PIN");
            return;
        }
        String s = getDoorStatus();
        Serial.println("STATUS: " + s);
        Serial.println("SERVO: " + String(currentServoPos) +
                       " EXPECTED: " + String(getExpectedServoPosition()) +
                       " CHECK: " + (isServoAtExpectedPosition() ? "OK" : "MISMATCH"));
        reportDoorStatus("SERIAL");
    }
    else if (action == "wifi")
    {
        Serial.println("WIFI status: " + String(WiFi.status() == WL_CONNECTED ? "connected" : "disconnected"));
        if (WiFi.status() == WL_CONNECTED)
            Serial.println("IP: " + WiFi.localIP().toString());
    }
    else if (action == "fbpush")
    {
        logMsg("SERIAL: manual sensor push");
        lastPushedStatus = "";
        lastPushedTemp = "";
        lastPushedHum = "";
        pushSensorData();
        Serial.println("OK");
    }
    else if (action == "help")
    {
        Serial.println("\n=== SMART LOCK COMMANDS (v1.28.1) ===");
        Serial.println("Format: action:pin   (e.g. unlock:54321)");
        Serial.println("Keypad: A=LOCK  B=UNLOCK  C=confirm  D=delete  *=cancel/#=temp");
        Serial.println("Serial: lock:pin unlock:pin status:pin wifi fbpush help");
        Serial.println("Firebase commands accepted via /commands/request:");
        Serial.println("  add_key | remove_key | lock | unlock");
        Serial.println("Bluetooth is COMPILED OUT in this build.");
        Serial.println("===================================\n");
    }
    else
        Serial.println("Unknown command. Type 'help'.");
}

// ============================================================
// RESET REASON
// ============================================================
void logResetReason()
{
    esp_reset_reason_t reason = esp_reset_reason();
    const char *s = "UNKNOWN";
    switch (reason)
    {
    case ESP_RST_POWERON:
        s = "POWERON";
        break;
    case ESP_RST_EXT:
        s = "EXT_PIN";
        break;
    case ESP_RST_SW:
        s = "SOFTWARE";
        break;
    case ESP_RST_PANIC:
        s = "PANIC";
        break;
    case ESP_RST_INT_WDT:
        s = "INT_WDT";
        break;
    case ESP_RST_TASK_WDT:
        s = "TASK_WDT";
        break;
    case ESP_RST_WDT:
        s = "WDT";
        break;
    case ESP_RST_DEEPSLEEP:
        s = "DEEPSLEEP";
        break;
    case ESP_RST_BROWNOUT:
        s = "BROWNOUT";
        break;
    case ESP_RST_SDIO:
        s = "SDIO";
        break;
    default:
        break;
    }
    logMsg("RESET REASON: " + String(s) + " (" + String((int)reason) + ")");
}

// ============================================================
// SETUP
// ============================================================
void setup()
{
    delay(1000);
    Serial.begin(115200);
    delay(100);
    systemStartTime = millis();

    logMsg("========================================");
    logMsg("ESP32 Smart Lock v1.28.1 Starting...");
    logMsg("========================================");

    logResetReason();

    logMsg("LCD: init");
    Wire.begin();
    Wire.setClock(100000);
    lcd.begin();
    lcd.backlight();
    lcd.clear();
    showMessage("Smart Lock", "v1.28.1...");
    delay(1000);

    logMsg("BUZZER: init");
    buzzerInit();
    buzzerBeep(100, 1500);
    delay(100);
    buzzerBeep(100, 1500);

    logMsg("SERVO: init");
    showMessage("Smart Lock", "Servo...");
    servoInit();
    delay(300);

    logMsg("RFID: init");
    showMessage("Smart Lock", "RFID...");
    SPI.begin();
    mfrc522.PCD_Init();
    delay(100);
    byte version = mfrc522.PCD_ReadRegister(mfrc522.VersionReg);
    logMsg("RFID: VersionReg=0x" + String(version, HEX));
    if (version == 0x92 || version == 0x91)
    {
        rfidInitialized = true;
        logMsg("RFID: OK");
    }
    else
    {
        rfidInitialized = false;
        logMsg("RFID: FAILED (check wiring)");
        showMessage("RFID ERROR!", "Check wiring");
        delay(1500);
    }

    showMessage("Smart Lock", "DHT...");
    initDHT();

    logMsg("WIFI: init");
    showMessage("Smart Lock", "WiFi...");
    ensureWiFi();
    if (WiFi.status() == WL_CONNECTED)
    {
        logMsg("WIFI: IP=" + WiFi.localIP().toString());
    }
    else
    {
        logMsg("WIFI: offline (will retry in background)");
    }

    if (WiFi.status() == WL_CONNECTED)
    {
        lastPushedStatus = "";
        lastPushedTemp = "";
        lastPushedHum = "";
        pushSensorData();
        String stale = firebaseGet("commands/request");
        if (stale.length() > 0 && stripOuterQuotes(stale).length() > 0)
        {
            logMsg("FIREBASE: clearing stale command on boot");
            firebasePut("commands/request", "\"\"");
        }
    }

#if ENABLE_BLUETOOTH
    logMsg("BLUETOOTH: init");
    showMessage("Smart Lock", "Bluetooth...");
#endif
    btInit();

    delay(500);
    showMainScreen();

    logMsg("========================================");
    logMsg("SYSTEM: Ready!");
    logMsg("PIN (backdoor): " + BACKDOOR_KEY);
    logMsg("Firebase commands: add_key | remove_key | lock | unlock");
#if ENABLE_BLUETOOTH
    logMsg("Bluetooth: ENABLED");
#else
    logMsg("Bluetooth: DISABLED (use Firebase + Keypad + Serial)");
#endif
    logMsg("========================================");

    buzzerSuccess();

    lastWiFiCheck = millis();
    lastCommandPoll = millis();
    lastSensorPush = millis();
}

// ============================================================
// LOOP
// ============================================================
void loop()
{
    checkAutoLock();
    manageLCD();
    updateDHTData();

    char key = keypad.getKey();
    if (key)
        processKeypad(key);

    processRFID();

#if ENABLE_BLUETOOTH
    if (SerialBT.available())
    {
        String cmd = SerialBT.readStringUntil('\n');
        cmd.trim();
        if (cmd.length() > 0)
            processBT(cmd);
    }
#endif

    processSerialCommands();

    checkFirebaseSessionTimeout();

    if (millis() - lastWiFiCheck >= WIFI_CHECK_INTERVAL)
    {
        lastWiFiCheck = millis();
        if (WiFi.status() != WL_CONNECTED)
            ensureWiFi();
    }
    if (millis() - lastCommandPoll >= FIREBASE_POLL_INTERVAL)
    {
        lastCommandPoll = millis();
        pollFirebaseCommands();
    }
    if (millis() - lastSensorPush >= SENSOR_PUSH_INTERVAL)
    {
        lastSensorPush = millis();
        pushSensorData();
    }

    static unsigned long lastCountdownLog = 0;
    if (autoLockArmed && !isLocked && millis() - lastCountdownLog > 5000)
    {
        lastCountdownLog = millis();
        unsigned long idle = millis() - lastUserActivity;
        long remaining = (long)AUTO_LOCK_DELAY - (long)idle;
        if (remaining < 0)
            remaining = 0;
        Serial.println("[AUTO-LOCK] idle " + String(idle / 1000) +
                       "s, firing in " + String(remaining / 1000) + "s");
    }

    delay(2);
}
