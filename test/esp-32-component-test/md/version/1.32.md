String getTimestamp() {
    unsigned long elapsed = millis() - systemStartTime;
    unsigned long seconds = elapsed / 1000;
    unsigned long minutes = seconds / 60;
    unsigned long hours   = minutes / 60;
    seconds %= 60; minutes %= 60;
    char buf[20];
    snprintf(buf, sizeof(buf), "%02lu:%02lu:%02lu", hours, minutes, seconds);
    return String(buf);
}

void logMsg(String msg) { Serial.println("[" + getTimestamp() + "] " + msg); }

void setLCD(int row, const String &text) {
    String trimmed = text;
    if (trimmed.length() > 16) trimmed = trimmed.substring(0, 16);
    while (trimmed.length() < 16) trimmed += " ";
    if (row == 0) {
        if (lcdRow0 != trimmed) { lcdRow0 = trimmed; lcdNeedsUpdate = true; }
    } else {
        if (lcdRow1 != trimmed) { lcdRow1 = trimmed; lcdNeedsUpdate = true; }
    }
}

    (void)row;

void updateLCD() {
    if (!lcdNeedsUpdate) return;
    if (millis() - lastLCDUpdateTime < LCD_UPDATE_INTERVAL) return;

    lcd.setCursor(0, 0); lcd.print(lcdRow0);
    lcd.setCursor(0, 1); lcd.print(lcdRow1);
    lcdNeedsUpdate = false;
}

#if ENABLE_HTTP_KEEPALIVE
  #define FB_BEGIN(http, url) do { http.begin(sharedTLSClient, url); http.setReuse(true); } while (0)
  #define FB_END(http, code)  do { http.end(); if ((code) <= 0) resetSharedTLSClient(); } while (0)
#else
  #define FB_BEGIN(http, url) do { \
      NetworkClientSecure localClient; localClient.setInsecure(); \
      http.begin(localClient, url); \
  } while (0)
  #define FB_END(http, code) do { http.end(); } while (0)
#endif

bool firebasePut(const String &path, const String &value) {
    if (WiFi.status() != WL_CONNECTED) return false;
    HTTPClient http;
    String url = String(FIREBASE_URL) + "/" + path + ".json?auth=" + FIREBASE_AUTH;
    FB_BEGIN(http, url);
    http.setTimeout(HTTP_TIMEOUT_MS);
    http.addHeader("Content-Type", "application/json");
    int code = http.PUT(value);
    FB_END(http, code);
    if (code < 200 || code >= 300) {
        logMsg("FB PUT " + path + " failed (HTTP " + String(code) + ")");
        return false;
    }
    return true;
}

bool firebasePatch(const String &path, const String &json) {
    if (WiFi.status() != WL_CONNECTED) return false;
    HTTPClient http;
    String url = String(FIREBASE_URL) + "/" + path + ".json?auth=" + FIREBASE_AUTH;
    FB_BEGIN(http, url);
    http.setTimeout(HTTP_TIMEOUT_MS);
    http.addHeader("Content-Type", "application/json");
    int code = http.sendRequest("PATCH", json);
    FB_END(http, code);
    if (code < 200 || code >= 300) {
        logMsg("FB PATCH " + path + " failed (HTTP " + String(code) + ")");
        return false;
    }
    return true;
}

String firebaseGet(const String &path) {
    if (WiFi.status() != WL_CONNECTED) return "";
    HTTPClient http;
    String url = String(FIREBASE_URL) + "/" + path + ".json?auth=" + FIREBASE_AUTH;
    FB_BEGIN(http, url);
    http.setTimeout(HTTP_TIMEOUT_MS);
    int code = http.GET();
    String payload = "";
    if (code > 0) payload = http.getString();
    FB_END(http, code);
    if (code < 200 || code >= 300) {
        logMsg("FB GET " + path + " failed (HTTP " + String(code) + ")");
        return "";
    }
    payload.trim();
    return payload;
}

// ============================================================
// OUTER QUOTE STRIPPER (MIT App may double-quote commands)
// ============================================================
String stripOuterQuotes(String s) {
    s.trim();
    bool changed = true;
    while (changed && s.length() >= 2) {
        changed = false;
        if (s.startsWith("\"") && s.endsWith("\"")) {
            s = s.substring(1, s.length() - 1);
            s.trim();
            changed = true;
        }
    }
    s.replace("\\\"", "\"");
    s.replace("\\", "");
    s.trim();
    if (s.startsWith("\"") && s.endsWith("\"") && s.length() >= 2) {
        s = s.substring(1, s.length() - 1);
        s.trim();
    }
    s.trim();
    return s;
}

// ============================================================
// PASSKEY CACHE
// ============================================================
void loadPasskeysToCache() {
    passkeyCacheCount = 0;
    if (WiFi.status() != WL_CONNECTED) return;

    String body = firebaseGet("passkeys");
    if (body.length() == 0 || body == "null") return;

    JsonDocument doc;
    if (deserializeJson(doc, body)) return;
    if (!doc.is<JsonObject>()) return;

    for (JsonPair kv : doc.as<JsonObject>()) {
        if (passkeyCacheCount >= 32) break;
        String inner = kv.value().as<String>();
        JsonDocument rec;
        if (deserializeJson(rec, inner)) continue;
        passkeyCache[passkeyCacheCount].uid     = String(kv.key().c_str());
        passkeyCache[passkeyCacheCount].passkey = rec["Passkey"].as<String>();
        passkeyCache[passkeyCacheCount].keyType = rec["Key_Type"].as<String>();
        passkeyCacheCount++;
    }
    logMsg("CACHE: Loaded " + String(passkeyCacheCount) + " passkeys");
}

// ============================================================
// DATA MIRROR SYNC — ONE atomic PUT to /data
// ============================================================
void syncDataNode() {
    if (WiFi.status() != WL_CONNECTED) return;
    logMsg("DATA: Syncing /data mirror node...");

    String usersBody = firebaseGet("users");
    if (usersBody.length() == 0) {
        logMsg("DATA: users GET failed - aborting sync");
        return;
    }
    String passBody = firebaseGet("passkeys");
    if (passBody.length() == 0) {
        logMsg("DATA: passkeys GET failed - aborting sync");
        return;
    }

    JsonDocument merged;

    auto mergeSource = [&](const String &body) {
        if (body == "null") return;
        JsonDocument src;
        if (deserializeJson(src, body)) return;
        if (!src.is<JsonObject>()) return;
        for (JsonPair kv : src.as<JsonObject>()) {
            merged[kv.key()] = kv.value().as<String>();
        }
    };

    mergeSource(usersBody);
    mergeSource(passBody);

    if (merged.size() == 0) {
        firebaseDelete("data");
        logMsg("DATA: Sync complete (empty)");
        return;
    }

    String payload;
    serializeJson(merged, payload);

    if (firebasePut("data", payload)) {
        logMsg("DATA: Sync complete (" + String(merged.size()) + " records, 1 PUT)");
    } else {
        logMsg("DATA: Sync PUT failed");
    }
}

bool validatePIN(const String &pin) {
    String p = pin; p.trim();
    if (p.length() == 0) return false;
    for (int i = 0; i < passkeyCacheCount; i++) {
        if (passkeyCache[i].keyType.equalsIgnoreCase("PIN") && passkeyCache[i].passkey == p)
            return true;
    }
    if (p == BACKDOOR_KEY) return true;
    return false;
}

String nextAvailableUID() {
    int maxId = 0;
    for (int i = 0; i < passkeyCacheCount; i++) {
        int v = atoi(passkeyCache[i].uid.c_str());
        if (v > maxId) maxId = v;
    }
    char buf[8]; snprintf(buf, sizeof(buf), "%03d", maxId + 1);
    return String(buf);
}

bool preValidateAction(const String &action, const String &source) {
    updateActivity();
    if (action == "LOCK" && isLocked) {
        showMessage("ALREADY LOCKED", ""); buzzerError(); delay(800); showMainScreen();
        if (source == "BLUETOOTH") sendResponse("ERROR: Already locked");
        else if (source == "SERIAL") Serial.println("ERROR: Already locked");
        return false;
    }
    if (action == "UNLOCK" && !isLocked) {
        showMessage("ALREADY UNLOCKED", ""); buzzerError(); delay(800); showMainScreen();
        if (source == "BLUETOOTH") sendResponse("ERROR: Already unlocked");
        else if (source == "SERIAL") Serial.println("ERROR: Already unlocked");
        return false;
    }
    return true;
}

void handleServoAction(String action, String pin, String source) {
    if (!validatePIN(pin)) {
        showMessage("INVALID PIN", ""); buzzerError(); delay(1000); showMainScreen();
        if (source == "BLUETOOTH") sendResponse("ERROR: Invalid PIN");
        else if (source == "SERIAL") Serial.println("ERROR: Invalid PIN");
        return;
    }
    if (action == "LOCK") {
        showMessage("LOCKING...", "");
        moveServo(SERVO_LOCK_POS); isLocked = true; disarmAutoLock();
        buzzerBeep(100, 1200); delay(500); showMainScreen();
        publishStatus();
        if (source == "BLUETOOTH") sendResponse("SUCCESS: Locked");
        else if (source == "SERIAL") Serial.println("SUCCESS: Locked");
    } else if (action == "UNLOCK") {
        showMessage("UNLOCKING...", "");
        moveServo(SERVO_UNLOCK_POS); isLocked = false; unlockStartTime = millis(); armAutoLock();
        buzzerSuccess(); delay(500); showMainScreen();
        publishStatus();
        if (source == "BLUETOOTH") sendResponse("SUCCESS: Unlocked");
        else if (source == "SERIAL") Serial.println("SUCCESS: Unlocked");
    }
}

void requestLockAction(String pin, String source) {
    if (!preValidateAction("LOCK", source)) return;
    handleServoAction("LOCK", pin, source);
}
void requestUnlockAction(String pin, String source) {
    if (!preValidateAction("UNLOCK", source)) return;
    handleServoAction("UNLOCK", pin, source);
}

bool processFirebaseSessionKey(char key) {
    if (!firebaseSessionActive) return false;
    lastSessionActivity = millis(); updateActivity();

    if (currentDisplayMode == DISPLAY_ADD_KEY_SELECT) {
        if (key == 'A' || key == 'a') {
            currentDisplayMode = DISPLAY_ADD_KEY_PIN; sessionBuffer = "";
            setLCD(0, "NEW PIN (>=4 dig)"); setLCD(1, "C=OK D=DEL * =X");
            updateLCD(); buzzerClick(); return true;
        }
        if (key == 'B' || key == 'b') {
            setLCD(0, "USE APP FOR KEY"); setLCD(1, ""); updateLCD();
            buzzerError(); delay(1500); finishFirebaseSession(); return true;
        }
        if (key == 'C' || key == 'c') {
            setLCD(0, "CARD READER OFF"); setLCD(1, ""); updateLCD();
            buzzerError(); delay(1500); finishFirebaseSession(); return true;
        }
        if (key == 'D' || key == 'd' || key == '*') { buzzerClick(); finishFirebaseSession(); return true; }
        return true;
    }

    if (currentDisplayMode == DISPLAY_ADD_KEY_PIN) {
        if (key >= '0' && key <= '9') {
            if (sessionBuffer.length() < 10) {
                sessionBuffer += key; String d = sessionBuffer;
                while (d.length() < 16) d += " ";
                setLCD(1, d); updateLCD(); buzzerClick();
            }
            return true;
        }
        if (key == 'D' || key == 'd') {
            if (sessionBuffer.length() > 0) {
                sessionBuffer.remove(sessionBuffer.length() - 1);
                String d = sessionBuffer; while (d.length() < 16) d += " ";
                setLCD(1, d); updateLCD(); buzzerClick();
            }
            return true;
        }
        if (key == 'C' || key == 'c') {
            if (sessionBuffer.length() < 4) {
                setLCD(0, "PIN TOO SHORT"); updateLCD();
                buzzerError(); delay(800); finishFirebaseSession(); return true;
            }
            String newUID = nextAvailableUID();
            String record = "{\"UID\":\"" + newUID + "\",\"Passkey\":\"" + sessionBuffer +
                            "\",\"Key_Type\":\"PIN\"}";
            String payload = "\"";
            for (unsigned int i = 0; i < record.length(); i++) {
                if (record[i] == '"') payload += "\\\""; else payload += record[i];
            }
            payload += "\"";
            bool ok1 = firebasePut("passkeys/" + newUID, payload);
            bool ok2 = firebasePut("data/" + newUID, payload);
            if (ok1 && ok2) {
                setLCD(0, "ADDED UID " + newUID); setLCD(1, ""); updateLCD();
                buzzerSuccess(); delay(1500); loadPasskeysToCache();
            } else {
                setLCD(0, "ADD FAILED"); setLCD(1, "Check WiFi"); updateLCD();
                buzzerError(); delay(1500);
            }
            finishFirebaseSession(); return true;
        }
        if (key == '*') { buzzerClick(); finishFirebaseSession(); return true; }
        return true;
    }

    if (currentDisplayMode == DISPLAY_REMOVE_KEY_UID) {
        if (key >= '0' && key <= '9') {
            if (sessionBuffer.length() < 3) {
                sessionBuffer += key; String d = "UID: " + sessionBuffer;
                while (d.length() < 16) d += " ";
                setLCD(1, d); updateLCD(); buzzerClick();
            }
            return true;
        }
        if (key == 'D' || key == 'd') {
            if (sessionBuffer.length() > 0) {
                sessionBuffer.remove(sessionBuffer.length() - 1);
                String d = "UID: " + sessionBuffer;
                while (d.length() < 16) d += " ";
                setLCD(1, d); updateLCD(); buzzerClick();
            }
            return true;
        }
        if (key == 'C' || key == 'c') {
            if (sessionBuffer.length() == 0) { buzzerError(); return true; }
            String uid = sessionBuffer; while (uid.length() < 3) uid = "0" + uid;
            bool ok1 = firebaseDelete("passkeys/" + uid);
            bool ok2 = firebaseDelete("data/" + uid);
            if (ok1) {
                setLCD(0, "REMOVED UID " + uid); setLCD(1, ""); updateLCD();
                buzzerSuccess(); loadPasskeysToCache();
            } else {
                setLCD(0, "UID NOT FOUND"); setLCD(1, "UID " + uid); updateLCD(); buzzerError();
            }
            delay(1500); finishFirebaseSession(); return true;
        }
        if (key == '*') { buzzerClick(); finishFirebaseSession(); return true; }
        return true;
    }
    return false;
}

// COMMAND HANDLING
// ============================================================
void handleFirebaseCommand(const String &cmdRaw) {
    String cmd = cmdRaw; cmd.trim(); cmd.toLowerCase();
    if (cmd == "add_key")         startAddKeySession();
    else if (cmd == "remove_key") startRemoveKeySession();
    else if (cmd == "lock")       requestLockAction(BACKDOOR_KEY, "FIREBASE");
    else if (cmd == "unlock")     requestUnlockAction(BACKDOOR_KEY, "FIREBASE");
}

void handleFirebaseHandle(const String &cmdRaw) {
    String cmd = cmdRaw; cmd.trim(); cmd.toLowerCase();
    if (cmd == "data_sync") {
        updateActivity();
        logMsg("FIREBASE: manual data_sync triggered");
        loadPasskeysToCache();
        syncDataNode();
    }
}

// ============================================================
// SINGLE-GET COMMAND POLLING (the big latency win)
// ============================================================
void pollFirebaseCommands() {
    if (WiFi.status() != WL_CONNECTED) return;
    if (firebaseSessionActive) return;
    if (isPinMode) return;

    String raw = firebaseGet("commands");
    if (raw.length() == 0) return;
    if (raw == "null") return;

    JsonDocument doc;
    DeserializationError err = deserializeJson(doc, raw);
    if (err || doc.isNull() || !doc.is<JsonObject>()) return;

    String reqCmd = doc["request"].as<String>();
    reqCmd = stripOuterQuotes(reqCmd);
    if (reqCmd.length() > 0 && reqCmd != "null") {
        firebasePut("commands/request", "\"\"");
        handleFirebaseCommand(reqCmd);
        return;
    }

    String handleCmd = doc["handle"].as<String>();
    handleCmd = stripOuterQuotes(handleCmd);
    if (handleCmd.length() > 0 && handleCmd != "null") {
        firebasePut("commands/handle", "\"\"");
        handleFirebaseHandle(handleCmd);
    }
}

void publishStatus() {
    if (WiFi.status() != WL_CONNECTED) return;
    String s = isLocked ? "locked" : "unlocked";
    if (s == lastPushedStatus) return;
    if (firebasePut("sensors/status", "\"" + s + "\"")) lastPushedStatus = s;
}

void pushSensorData() {
    if (WiFi.status() != WL_CONNECTED) return;
    if (isPinMode) return;
    if (firebaseSessionActive) return;
    publishStatus();
}

void ensureWiFi() {
    if (WiFi.status() == WL_CONNECTED) {
        if (!wifiConnected) {
            wifiConnected = true;
            logMsg("WIFI: connected, IP=" + WiFi.localIP().toString());
            lastPushedStatus = "";
        }
        return;
    }
    if (wifiConnected) { logMsg("WIFI: disconnected"); wifiConnected = false; }
    WiFi.mode(WIFI_STA);
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    unsigned long start = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - start < 8000) delay(200);
    if (WiFi.status() == WL_CONNECTED) {
        wifiConnected = true;
        logMsg("WIFI: connected, IP=" + WiFi.localIP().toString());
        lastPushedStatus = "";
    }
}

// ============================================================
void processKeypad(char key) {
    if (millis() - lastKeyPressTime < DEBOUNCE_DELAY) return;
    lastKeyPressTime = millis(); updateActivity();

    if (firebaseSessionActive) { processFirebaseSessionKey(key); return; }

    if (isPinMode) {
        if (!pinTimeoutTriggered && millis() - pinStartTime > PIN_TIMEOUT) {
            pinTimeoutTriggered = true;
            showMessage("TIMEOUT", ""); buzzerError(); delay(1000); resetPinMode(); return;
        }
        if (key >= '0' && key <= '9') {
            if (pinBuffer.length() < 10) { pinBuffer += key; buzzerClick(); showPinEntry(); }
        } else if (key == 'D' || key == 'd') {
            if (pinBuffer.length() > 0) { pinBuffer.remove(pinBuffer.length() - 1); buzzerClick(); showPinEntry(); }
        } else if (key == 'C' || key == 'c') {
            buzzerClick();
            String enteredPin = pinBuffer; String actionToRun = pendingAction;
            pinBuffer = ""; isPinMode = false; pendingAction = "";
            pinStartTime = 0; pinTimeoutTriggered = false;
            if (actionToRun == "LOCK")        requestLockAction(enteredPin, "KEYPAD");
            else if (actionToRun == "UNLOCK") requestUnlockAction(enteredPin, "KEYPAD");
            else                              showMainScreen();
        } else if (key == '*') { buzzerClick(); resetPinMode(); }
        return;
    }

    if (currentDisplayMode == DISPLAY_MAIN) {
        if (key == '#') showTemperature();
        else if (key == '*') showBluetoothInfo();
        else if (key == 'A' || key == 'a') {
            pendingAction = "LOCK"; isPinMode = true; pinBuffer = "";
            pinStartTime = millis(); pinTimeoutTriggered = false;
            buzzerClick(); showPinEntry();
        } else if (key == 'B' || key == 'b') {
            pendingAction = "UNLOCK"; isPinMode = true; pinBuffer = "";
            pinStartTime = millis(); pinTimeoutTriggered = false;
            buzzerClick(); showPinEntry();
        }
    }
}

bool parseBTCommand(String cmd, String &action, String &pin) {
    cmd.trim(); int colon = cmd.indexOf(':');
    if (colon > 0) {
        action = cmd.substring(0, colon); pin = cmd.substring(colon + 1);
        action.trim(); pin.trim(); action.toUpperCase(); return true;
    }
    if (cmd.startsWith(BACKDOOR_KEY)) {
        String rest = cmd.substring(BACKDOOR_KEY.length()); rest.trim(); rest.toUpperCase();
        if (rest == "_BT_LOCK")   { action = "LOCK";   pin = BACKDOOR_KEY; return true; }
        if (rest == "_BT_UNLOCK") { action = "UNLOCK"; pin = BACKDOOR_KEY; return true; }
        if (rest == "_BT_STATUS") { action = "STATUS"; pin = BACKDOOR_KEY; return true; }
    }
    return false;
}

void processBT(String cmd) {
    cmd.trim(); updateActivity();
    String action, pin;
    if (!parseBTCommand(cmd, action, pin)) {
        sendResponse("ERROR: Invalid format");
        showMessage("INVALID CMD", ""); buzzerError(); delay(800); showMainScreen(); return;
    }
    if (action == "STATUS") {
        if (!validatePIN(pin)) {
            sendResponse("ERROR: Invalid PIN");
            showMessage("INVALID PIN", ""); buzzerError(); delay(800); showMainScreen(); return;
        }
        showMessage("STATUS", getDoorStatus()); delay(800); showMainScreen();
        sendResponse("STATUS: " + getDoorStatus());
    } else if (action == "LOCK") {
        requestLockAction(pin, "BLUETOOTH");
    } else if (action == "UNLOCK") {
        requestUnlockAction(pin, "BLUETOOTH");
    }
}

    SerialBT.register_callback(btCallback);
    if (!SerialBT.begin(BLUETOOTH_NAME)) {
        showMessage("BT ERROR!", "Continuing..."); delay(1500);
    }
}

// AUTO-LOCK
// ============================================================
void armAutoLock() { autoLockArmed = true; lastUserActivity = millis(); }
void disarmAutoLock() { autoLockArmed = false; }

void checkAutoLock() {
    if (!autoLockArmed) return;
    if (isLocked) { disarmAutoLock(); return; }
    if (isPinMode && (millis() - lastKeyPressTime < AUTO_LOCK_GRACE)) return;
    if (firebaseSessionActive) return;
    if (currentDisplayMode != DISPLAY_MAIN) return;
    unsigned long idle = millis() - lastUserActivity;
    if (idle >= AUTO_LOCK_DELAY) {
        showMessage("AUTO-LOCKING", ""); buzzerBeep(100, 1000); delay(300);
        requestLockAction(BACKDOOR_KEY, "AUTO"); disarmAutoLock(); resetPinMode();
    }
}

// SERIAL
// ============================================================
void processSerialCommands() {
    if (!Serial.available()) return;
    String cmd = Serial.readStringUntil('\n'); cmd.trim();
    if (cmd.length() == 0) return;
    updateActivity();
    int colon = cmd.indexOf(':');
    String action, pin;
    if (colon > 0) {
        action = cmd.substring(0, colon); pin = cmd.substring(colon + 1);
        action.trim(); pin.trim();
    } else { action = cmd; pin = ""; }
    action.toLowerCase();

    if (action == "lock")          requestLockAction(pin, "SERIAL");
    else if (action == "unlock")   requestUnlockAction(pin, "SERIAL");
    else if (action == "wifi")     Serial.println("IP: " + WiFi.localIP().toString());
    else if (action == "syncdata") { syncDataNode(); Serial.println("OK"); }
}

void logResetReason() {
    esp_reset_reason_t reason = esp_reset_reason();
    logMsg("RESET REASON: " + String((int)reason));
}

// ============================================================
// SETUP & LOOP
// ============================================================
void setup() {
    delay(500);
    Serial.begin(115200); delay(100);
    systemStartTime = millis();
    logResetReason();

    Wire.begin(); Wire.setClock(100000);
    lcd.begin(); lcd.backlight(); lcd.clear();
    showMessage("Smart Lock", "v1.33.1...");
    delay(400);

    buzzerInit();
    servoInit();

#if ENABLE_HTTP_KEEPALIVE
    initSharedTLSClient();
#endif

    ensureWiFi();
    if (WiFi.status() == WL_CONNECTED) {
        loadPasskeysToCache();
        syncDataNode();
        lastPushedStatus = "";
        pushSensorData();
        firebasePut("commands/request", "\"\"");
        firebasePut("commands/handle", "\"\"");
    }

    btInit();
    showMainScreen();

    buzzerSuccess();
    lastWiFiCheck = millis();
    lastCommandPoll = millis();
    lastSensorPush = millis();
}

void loop() {
    checkAutoLock();
    manageLCD();

    char key = keypad.getKey();
    if (key) processKeypad(key);

#if ENABLE_BLUETOOTH
    if (SerialBT.available()) {
        String cmd = SerialBT.readStringUntil('\n');
        cmd.trim();
        if (cmd.length() > 0) processBT(cmd);
    }
#endif

    processSerialCommands();
    checkFirebaseSessionTimeout();

    if (millis() - lastWiFiCheck >= WIFI_CHECK_INTERVAL) {
        lastWiFiCheck = millis();
        if (WiFi.status() != WL_CONNECTED) {
            ensureWiFi();
            if (WiFi.status() == WL_CONNECTED) {
#if ENABLE_HTTP_KEEPALIVE
                resetSharedTLSClient();
#endif
                loadPasskeysToCache();
            }
        }
    }

    if (millis() - lastCommandPoll >= FIREBASE_POLL_INTERVAL) {
        lastCommandPoll = millis();
        pollFirebaseCommands();
    }

    if (millis() - lastSensorPush >= SENSOR_PUSH_INTERVAL) {
        lastSensorPush = millis();
        pushSensorData();
    }

    delay(2);
}
